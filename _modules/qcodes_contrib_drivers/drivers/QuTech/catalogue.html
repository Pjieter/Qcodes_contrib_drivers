"""Module types, rack slots, front-panel ports and knob settings for the IVVI rack.

Other tools can read this catalogue without importing the drivers.
Entries with missing module pages infer their ports from the module index,
as marked in the comments.
See https://qtwork.tudelft.nl/~schouten/ivvi/doc-mod/index-mod.htm.
"""

from dataclasses import asdict, dataclass
from typing import Literal

PortKind = Literal["drive", "sense", "current", "gate", "signal"]
SLOTS = {"source": ("Sa", "Sb", "Sc", "Sd"), "measure": ("Ma", "Mb")}
_PKG = "qcodes_contrib_drivers.drivers.QuTech"
_DOC = "https://qtwork.tudelft.nl/~schouten/ivvi/doc-mod/"
_DEFAULT_BACKPLANE = ("<default>",)

@dataclass(frozen=True)
class ModuleSpec:
    """Catalogue metadata for a virtual IVVI module.

    ports_verified is False when the ports are inferred from the module index.
    backplane lists the port IDs wired to a rack iso-amp or DAC.
    S5 pulse modules have no backplane signal route by default.
    A combined module with its own ISO out sits in a source slot, and its
    output takes the slot's DAC, as the S3b or S4c board inside it does.
    """

    type: str
    kind: Literal["source", "measure", "combined", "misc"]
    slot_kind: Literal["source", "measure", "any"]
    label: str
    ports: tuple[Port, ...]
    knobs: tuple[Knob, ...]
    doc_url: str
    driver: str
    ports_verified: bool = True
    backplane: tuple[str, ...] = _DEFAULT_BACKPLANE

    def __post_init__(self) -> None:
        if self.backplane is _DEFAULT_BACKPLANE:
            if self.kind == "measure":
                ports = tuple(p.id for p in self.ports if p.kind in {"sense", "current"})
            elif (self.kind == "source" and not self.type.startswith("S5")) or (
                self.kind == "combined" and any(p.id == "iso_out" for p in self.ports)
            ):
                ports = tuple(
                    p.id
                    for p in self.ports
                    if p.kind in {"drive", "gate"}
                    and p.id in {"out", "out+", "out-", "outA", "outB"}
                )
            else:
                ports = ()
            object.__setattr__(self, "backplane", ports)
        unknown = set(self.backplane) - {p.id for p in self.ports}
        if unknown:
            raise ValueError(f"{self.type}: unknown backplane ports {sorted(unknown)}")

def slots_for(slot_kind: str) -> tuple[str, ...]:
    """Return the rack slots that accept this slot kind."""
    return (
        SLOTS["source"] + SLOTS["measure"] if slot_kind == "any" else SLOTS[slot_kind]
    )

def catalogue_json() -> dict[str, dict]:
    """Return the catalogue with ports as triples and settings as JSON-safe lists."""
    out = {}
    for module_type, spec in CATALOGUE.items():
        entry = asdict(spec)
        entry["ports"] = [[port.id, port.label, port.kind] for port in spec.ports]
        entry["backplane"] = list(spec.backplane)
        entry["knobs"] = [
            {**asdict(knob), "values": list(knob.values)} for knob in spec.knobs
        ]
        del entry["type"]
        out[module_type] = entry
    return out
